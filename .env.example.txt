# .env.example — шаблон конфигурации для локального запуска
# Проект: «Ремонт под контролем»
# Не содержит реальных секретов — только структура и значения по умолчанию

# ==================== SERVER ====================
SERVER_PORT=8080
SERVER_CONTEXT_PATH=/api

# ==================== DATABASE ====================
# PostgreSQL 16
DB_HOST=localhost
DB_PORT=5432
DB_NAME=repair_control
DB_USER=postgres
DB_PASSWORD=postgres

# ==================== JPA / HIBERNATE ====================
# Для локальной разработки обычно используется create-drop или validate.
# В production — только validate или none.
SPRING_JPA_HIBERNATE_DDL_AUTO=validate
SPRING_JPA_SHOW_SQL=true
SPRING_JPA_PROPERTIES_HIBERNATE_FORMAT_SQL=true

# ==================== FLYWAY ====================
# Миграции выполняются автоматически при запуске.
# При необходимости можно отключить: SPRING_FLYWAY_ENABLED=false
SPRING_FLYWAY_ENABLED=true
SPRING_FLYWAY_LOCATIONS=classpath:db/migration

# ==================== SECURITY / JWT ====================
# Секретный ключ для подписи JWT-токенов.
# В локальной среде допустимо значение по умолчанию.
# В production ОБЯЗАТЕЛЬНО заменить на реальный секрет.
JWT_SECRET=change-this-secret-in-production
JWT_EXPIRATION_MS=86400000

# ==================== FILE STORAGE ====================
# Директория для загружаемых файлов.
# Путь указывается относительно корня проекта.
FILE_STORAGE_PATH=./uploads

# Максимальные размеры файлов (в байтах)
# Фото: до 10 МБ = 10_485_760 байт
MAX_FILE_SIZE_PHOTO=10485760
# Видео: до 50 МБ = 52_428_800 байт
MAX_FILE_SIZE_VIDEO=52428800
# Максимальное количество фотографий на одну подачу
MAX_PHOTO_COUNT=10
# Минимальное количество фотографий на одну подачу
MIN_PHOTO_COUNT=1

# ==================== CORS ====================
# Разрешённый origin для frontend (локальная разработка)
CORS_ALLOWED_ORIGINS=http://localhost:3000,http://localhost:5173

# ==================== LOGGING ====================
# Уровень логирования
LOGGING_LEVEL_ROOT=INFO
LOGGING_LEVEL_COM_REPAIRCONTROL=DEBUG
